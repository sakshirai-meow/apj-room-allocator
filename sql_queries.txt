CREATE DATABASE college_timetable;
USE college_timetable;

CREATE TABLE timetable (
    room VARCHAR(10),
    academic_year VARCHAR(10),
    day VARCHAR(10),
    slot VARCHAR(20),
    class_info TEXT
);

CREATE TABLE class_cancellations (
    room VARCHAR(10),
    date DATE,
    slot VARCHAR(20),
    class_info TEXT,
    PRIMARY KEY (room, date, slot)
);

CREATE TABLE bookings (
    room VARCHAR(10),
    date DATE,
    slot VARCHAR(20),
    purpose TEXT,
    booked_by VARCHAR(100),
    booker_contactno VARCHAR(20),
    PRIMARY KEY (room, date, slot)
);

ALTER TABLE college_timetable.bookings
ADD COLUMN status VARCHAR(20) DEFAULT 'confirmed';

CREATE TABLE weekly_schedule (
    room VARCHAR(10),
    date DATE,
    slot VARCHAR(20),
    class_info TEXT,
    status VARCHAR(20),
    booked_by VARCHAR(100),
    booker_contactno VARCHAR(20),
    booking_purpose TEXT,
    PRIMARY KEY (room, date, slot)
);
